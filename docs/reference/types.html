types

server

typedef struct server {
    short unsigned int domain;
    int service;
    int protocol;
    short unsigned int port;
    int backlog;
    uint32_t interface;
    int socket;
    struct sockaddr_in address;
} server;

http

typedef struct http {
    server *srv;
    dict *routes;
    char static_dir[256];
    unsigned short port;
    size_t thread_cnt;
    void (*launch)(struct http *http_server);
} http;

request

typedef struct request {
    dict *body;
    dict *req_line;
    dict *headers;
} request;

route

typedef struct route {
    int methods[METHOD_COUNT];
    char *uri;
    char *(*route_func)(http *srv, request *req);
} route;

executor

typedef void (*exec_func)(void *arg);

typedef struct exec_work {
    exec_func func;
    void *arg;
} exec_work;

typedef struct executor {
    queue *queue;
    pthread_t *threads;
    pthread_mutex_t mutex;
    pthread_cond_t non_empty;
    pthread_cond_t empty;
    size_t working_count;
    size_t pending_count;
    size_t thread_count;
    bool stop;
} executor;

dict

typedef struct dict {
    entry *entries[HASH_SIZE];  /* HASH_SIZE = 69 */
} dict;

struct entry {
    char *key;
    void *value;
    entry *next;
};
